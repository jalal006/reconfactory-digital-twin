cmake_minimum_required(VERSION 3.16)
project(reconfactory_gazebo_watchdog)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(nlohmann_json REQUIRED)
add_library(reconfactory-actuator-watchdog SHARED src/ActuatorWatchdog.cc)
target_compile_features(reconfactory-actuator-watchdog PUBLIC cxx_std_17)
target_compile_options(reconfactory-actuator-watchdog PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(reconfactory-actuator-watchdog PRIVATE include)
target_link_libraries(reconfactory-actuator-watchdog gz-sim8::gz-sim8
  gz-transport13::gz-transport13 gz-plugin2::register nlohmann_json::nlohmann_json)
install(TARGETS reconfactory-actuator-watchdog LIBRARY DESTINATION lib)
ament_environment_hooks(hooks/plugin_path.dsv.in)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(watchdog_policy_test test/policy_test.cc)
  target_compile_features(watchdog_policy_test PUBLIC cxx_std_17)
  target_include_directories(watchdog_policy_test PRIVATE include)
endif()
ament_package()
